  1 #include<stdio.h>
  2 int main()
  3 {
  4     int n,graph[10][10],visited[10]={0};
  5     int queue[10],front=0,rear=-1;
  6     printf("Enter number of nodes: ");
  7     scanf("%d",&n);
  8     printf("Enter adjacency matrix:\n");
  9     for(int i=0;i<n;i++)
 10         for(int j=0;j<n;j++)
 11             scanf("%d",&graph[i][j]);
 12      int start;
 13      printf("Enter starting node: ");
 14      scanf("%d",&start);
 15
 16      visited[start]=1;
 17      queue[++rear]=start;
 18      printf("BFS order: ");
 19
 20      while(front<=rear)
 21      {
 22          int node=queue[front++];
 23          printf("%d",node);
 24          for(int i=0;i<n;i++)
 25          {
 26              if(graph[node][i]==1 && visited[i]==0)
 27              {
 28                  visited[i]=1;
 29                  queue[++rear]=i;
 30              }
 31          }
 32      }
 33      return 0;
