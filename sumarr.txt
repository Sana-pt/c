#include<stdio.h>
 int sumarr(int arr[],int n)
 {
     int i,sum=0;
     for(i=0;i<=n;i++)
      {
          sum=sum+arr[i];
      }
      return sum;
 }
 int main()
  {
    int arr[100],n,i;
    printf("Enter no.of elements: ");
    scanf("%d",&n);
    printf("Enter array elements:\n");
    for(i=0;i<n;i++)
    {
        scanf("%d",&arr[i]);
    }
    printf("sum=%d",sumarr(arr,n));
    return 0;
 }
